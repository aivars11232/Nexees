SPECIFICATION:
    ID: specification.arch_interfaces
    NAME: "Interfaces between subsystems"
    VERSION: "0.5.0"
    KIND: kind.part.data
    DESCRIPTION: "The boundaries across which subsystems interact, with the invariants every implementation keeps. This part fixes what crosses each boundary and under which rules; the concrete types, wire schemas and signatures are fixed by the tasks listed in schema_tasks."

DATA:
    ID: data.arch_if_attach
    TYPE: OBJECT[REF(type.arch_interface)]
    VALUE:
        interface_id: "IF-ATTACH"
        name: "UI client attachment"
        providers: ["SS-LOCAL-HOST"]
        consumers: ["SS-DESKTOP-CLIENT", "SS-ANDROID-CLIENT"]
        carries: "Typed user intents from a UI client to its own device's local host, and the client's own view state for the host to keep; state snapshots and domain events from the host back to the client, and the view state it keeps for that client."
        invariants: ["Uses the shared protocol message types; the client never touches the state store, tools or secrets directly.", "On Desktop the window and the host are separate processes joined by local IPC that authenticates the peer process by operating-system identity, admits only the same OS user, and bounds message sizes, queues and paths (RC-23).", "An intent that causes an effect is authorized by the host's permission engine like any other request; a UI convenience is never authority.", "Detaching or closing a client never stops the host or authorized background work, and reattaching never starts a second executor (RC-04).", "A client's view state is presentation only: the host checks its shape, keeps it for the client on that channel, and reads no intent, target or binding from it (ST-VIEW, AD-06). A client and a host that agreed on a protocol version without these messages do without them."]
        schema_tasks: ["TASK-007", "TASK-009", "TASK-012", "TASK-057"]

        providers: ["SS-REMOTE"]
        consumers: ["SS-REMOTE"]

        schema_tasks: ["TASK-006", "TASK-007", "TASK-070"]
