SPECIFICATION:
    ID: specification.arch_state
    NAME: "Data ownership and authoritative state"
    VERSION: "0.5.0"
    KIND: kind.part.data
    DESCRIPTION: "Which state exists, which single subsystem writes it, where its authoritative copy lives, what other copies may exist and whether it may travel to another device. Every category has exactly one owner; the owner's subsystem record lists it in owns_state. Derived state is rebuilt from authoritative state and is never an input to authority."

DATA:
    ID: data.arch_state_content
    TYPE: OBJECT[REF(type.arch_state)]
    VALUE:
        state_id: "ST-CONTENT"
        name: "Workspace content"
        content: "The source files, LCL specification files and other files of a workspace replica, with their content revisions."
        owner: "SS-WORKSPACES"
        store: "The device-local replica under that device's authorized root, written only through workspace scope, atomic writes and the revision service, whoever the writer is: the user's editor, an agent tool, sync or import publication."
        authority_class: "authoritative"
        copies: "Another device's replica of a selected workspace is an independent replica reconciled by sync. Remote-only listings and unsaved editor buffers are views, not content."
        sync_policy: "selected_content"
        sensitivity: "private"
        notes: "Each replica is authoritative for edits made on its device; concurrent edits become a sync conflict, never a silent replacement (A6). Unsaved buffers are protected from incoming changes (A3)."
        schema_tasks: ["TASK-006", "TASK-013", "TASK-069"]
